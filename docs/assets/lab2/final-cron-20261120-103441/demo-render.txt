apiVersion: batch/v1

metadata:
  labels:
    app.kubernetes.io/component: report
    app.kubernetes.io/name: venue-booking
  name: venue-report-daily
  namespace: venue-booking-final
spec:
  concurrencyPolicy: Forbid
  failedJobsHistoryLimit: 2
  jobTemplate:
    metadata:
      labels:
        app.kubernetes.io/component: report
        app.kubernetes.io/name: venue-booking
    spec:
      activeDeadlineSeconds: 1800
      backoffLimit: 1
      completions: 1
      parallelism: 1
      template:
        metadata:
          labels:
            app.kubernetes.io/component: report
            app.kubernetes.io/name: venue-booking
        spec:
          automountServiceAccountToken: false
          containers:
          - args:
            - --spring.main.web-application-type=none
            - --report.run=true
            env:
            - name: REPORT_JOB_ID
              valueFrom:
                fieldRef:
                  fieldPath: metadata.labels['batch.kubernetes.io/job-name']
            - name: REPORT_SOURCE
              value: SCHEDULED
            - name: REPORT_WINDOW_MODE
              value: TODAY

              valueFrom:
                configMapKeyRef:
                  key: SPRING_DATASOURCE_URL
                  name: venue-config
            - name: SPRING_DATASOURCE_USERNAME
              valueFrom:
                secretKeyRef:
                  key: POSTGRES_USER
                  name: venue-db-credentials

              valueFrom:
                secretKeyRef:
                  key: POSTGRES_PASSWORD
                  name: venue-db-credentials
            image: venue-booking-api:lab2-20261113
            imagePullPolicy: IfNotPresent
            name: report
            resources:
              limits:
                cpu: 500m
                memory: 512Mi
              requests:
                cpu: 100m
                memory: 256Mi
          restartPolicy: Never
  schedule: '*/2 * * * *'
  startingDeadlineSeconds: 600
  successfulJobsHistoryLimit: 3
