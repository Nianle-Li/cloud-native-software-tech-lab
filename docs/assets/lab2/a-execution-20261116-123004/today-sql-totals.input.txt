SELECT json_agg(row_to_json(t) ORDER BY venue_id) FROM (SELECT v.id AS venue_id,count(r.id) AS reservation_count,coalesce(sum(extract(epoch from(r.end_time-r.start_time))/60),0)::bigint AS reserved_minutes FROM venues v LEFT JOIN reservations r ON r.venue_id=v.id AND r.status='ACTIVE' AND r.reservation_date=DATE '2026-11-16' GROUP BY v.id) t;
